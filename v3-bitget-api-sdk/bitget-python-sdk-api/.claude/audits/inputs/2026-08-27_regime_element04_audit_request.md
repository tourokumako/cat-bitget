```python
  156  def _uptrend_state_5m(df5: pd.DataFrame) -> np.ndarray:
  157      """上げトレンド状態機械（RG-v4 新設・RG-v5 で複数足化・過去バーのみ＝因果的・上げ専用）。

  167        15分構造破断＝close < 直近 TREND_STRUCT_REL 本の確定15分安値の最小（上げ構造の終了を速く検出）。"""

  169      n = len(c)

  171      o15 = df5["open"].resample("15min", label="right", closed="right").first()
  172      h15 = df5["high"].resample("15min", label="right", closed="right").max()

  180            ((t15["c"] / t15["c"].shift(TREND_MOM15_K) - 1) >= TREND_MOM15_AMP)

  184

  186      s = 0; H = L = 0.0; last_new = 0
  187      rel_i = 0; H_prev = np.inf; L_prev = 0.0; rel_time = -10**9
  188      dq: deque = deque()          # (idx, close) の単調 min-deque＝窓内最小を O(1) で
  189      for i in range(n):
  190          while dq and dq[-1][1] >= c[i]:
  191              dq.pop()
  192          dq.append((i, c[i]))
  193          lo90 = max(i - 18, rel_i); lo6 = max(i - 72, rel_i)
  194          while dq and dq[0][0] < lo6:
  195              dq.popleft()
  196          m6 = min(v for _, v in dq)
  197          m90 = min((v for j, v in dq if j >= lo90), default=c[i])
  198          if s == 0:
  199              trig = ((c[i] / m90 - 1) >= TREND_T90 or (c[i] / m6 - 1) >= TREND_T6
  200                      or A15[i] or D15[i])
  201              cont = (i - rel_time) <= TREND_CONT_W and c[i] > H_prev * (1 + TREND_CONT_M)
  202              if trig or cont:
  203                  s = 1; L = m6 if trig else L_prev; H = c[i]; last_new = i
  204          else:
  205              if c[i] > H:
  206                  H = c[i]; last_new = i
  207              if (c[i] <= H - TREND_RETR * (H - L) or (i - last_new) >= TREND_STALL
  208                      or (np.isfinite(l15_rel[i]) and c[i] < l15_rel[i])):
  209                  s = 0; rel_i = i; rel_time = i; H_prev = H; L_prev = L
  210          st[i] = bool(s)
  211      return st
```
